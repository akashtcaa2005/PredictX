# ============================================================================
# Polybot PAPER trading engine configuration.
#
#   cp env.paper.example.txt env.paper.txt
#
# This file must NEVER contain a wallet key. The paper engine deletes every
# wallet credential from its process environment before it does anything else
# (polybot.config.settings.scrub_wallet_secrets), so even if one were present
# here it would be discarded — but keep this file clean regardless.
#
# The paper engine reads REAL order books from the public CLOB endpoints and
# matches orders internally against that real depth, with virtual money. It
# writes only to data/paper/.
# ============================================================================

# --- markets (required) -----------------------------------------------------
# Comma-separated Polymarket condition ids. Use the same ids you intend to trade
# live so paper results are comparable.
MARKETS=

# --- virtual account --------------------------------------------------------
PAPER_STARTING_CASH=1000

# How realistic the simulation is:
#   FILL_LATENCY_MS   an order cannot fill until this long after submission
#   SLIPPAGE_TICKS    price penalty applied to every fill, in tick sizes
#   FEE_BPS           fee charged on notional (Polymarket is 0 on most markets)
#   MAKER_FILL_RATIO  fraction of the remaining size that fills per pass — below
#                     1.0 produces partial fills, which is closer to reality for
#                     a resting order
PAPER_FILL_LATENCY_MS=250
PAPER_SLIPPAGE_TICKS=0
PAPER_FEE_BPS=0
PAPER_MAKER_FILL_RATIO=1.0

# --- order sizing (shared with the strategy framework) ----------------------
MAX_TRADE_VALUE=5

# --- risk limits ------------------------------------------------------------
