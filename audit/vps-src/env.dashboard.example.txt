# ============================================================================
# Polybot dashboard configuration.
#
#   cp env.dashboard.example.txt env.dashboard.txt
#   chmod 600 env.dashboard.txt
#
# The dashboard reads this file plus the process environment (real environment
# variables win). Nothing here is a secret about your wallet — the dashboard
# reads live account state through env.txt, which it never copies or exposes.
# ============================================================================

# --- required ---------------------------------------------------------------
# Signs nothing by itself, but is required to exist and be long. Generate with:
#   python run_dashboard.py --gen-secret
DASHBOARD_SECRET_KEY=

# --- http -------------------------------------------------------------------
# Bind loopback only and let nginx terminate TLS (the recommended setup).
# To serve TLS directly instead, set DASHBOARD_HOST=0.0.0.0 and the TLS paths.
DASHBOARD_HOST=127.0.0.1
DASHBOARD_PORT=8787
DASHBOARD_BEHIND_PROXY=true

# Direct TLS (leave blank when nginx terminates):
# DASHBOARD_TLS_CERT=/etc/letsencrypt/live/example.com/fullchain.pem
# DASHBOARD_TLS_KEY=/etc/letsencrypt/live/example.com/privkey.pem

# Your public URL. Used to validate websocket Origin headers.
DASHBOARD_PUBLIC_URL=https://dashboard.example.com
# DASHBOARD_ALLOWED_ORIGINS=https://dashboard.example.com,https://alt.example.com

# --- security ---------------------------------------------------------------
# Set false ONLY if you access the dashboard over plain http on a private
# network; leaving it true over http means the browser will drop your session.
DASHBOARD_COOKIE_SECURE=true
DASHBOARD_SESSION_TTL_S=43200
DASHBOARD_SESSION_IDLE_S=7200
DASHBOARD_LOGIN_MAX_ATTEMPTS=6
DASHBOARD_LOGIN_LOCKOUT_S=300

# The first admin is created on first start. If you leave the password blank a
# strong one is generated and printed ONCE to the service journal.
DASHBOARD_ADMIN_USER=admin
# DASHBOARD_ADMIN_PASSWORD=

# --- service control --------------------------------------------------------
# Set false to make the dashboard strictly read-only (no start/stop buttons).
DASHBOARD_CONTROL_ENABLED=true
DASHBOARD_USE_SUDO=true
SYSTEMCTL_BIN=/usr/bin/systemctl
PAPER_UNIT=polybot-paper.service
LIVE_UNIT=polybot-live.service
DASHBOARD_UNIT=polybot-dashboard.service

# --- live account sync ------------------------------------------------------
# Reads wallet, orders, fills and positions straight from Polymarket using the
# credentials in env.txt. Strictly read-only: the dashboard cannot place orders.
DASHBOARD_LIVE_SYNC=true
LIVE_SYNC_INTERVAL_S=3
LIVE_SLOW_SYNC_INTERVAL_S=30

# --- market data ------------------------------------------------------------
DASHBOARD_ASSETS=BTC,ETH,SOL,XRP
DASHBOARD_DURATIONS=5m,15m
MARKET_DISCOVERY_INTERVAL_S=60
MAX_MARKETS_TRACKED=24
# Candle width for the indicator series (seconds). 15s over a 5m market gives
# 20 bars per market cycle; raise it if you want smoother, slower indicators.
CANDLE_INTERVAL_S=15
CANDLE_RETENTION=5000

# --- streaming --------------------------------------------------------------
BROADCAST_INTERVAL_S=1.0
VPS_INTERVAL_S=5
HISTORY_POINTS=600

# --- alert thresholds -------------------------------------------------------
ALERT_CPU_PCT=85
ALERT_MEMORY_PCT=88
ALERT_DISK_PCT=90
ALERT_LOW_BALANCE=5
ALERT_ENGINE_STALE_S=60
ALERT_WS_STALE_S=60
# The compiled strategy does not publish TP/SL levels, so these define what the
# dashboard reports as "profit target hit" / "stop loss hit": the realised PnL
# of a completed round trip, in collateral units.
ALERT_PROFIT_TARGET=2.0
ALERT_STOP_LOSS=2.0
# Repeat suppression for a condition that stays true.
ALERT_COOLDOWN_S=300

# --- notification channels --------------------------------------------------
# Browser notifications need no configuration (enable them in the Alerts tab).
# Only alerts at or above this severity go to email/telegram/discord.
NOTIFY_MIN_SEVERITY=warning

EMAIL_ENABLED=false
SMTP_HOST=
SMTP_PORT=587
SMTP_USER=
SMTP_PASSWORD=
SMTP_STARTTLS=true
EMAIL_FROM=polybot@example.com
EMAIL_TO=you@example.com

TELEGRAM_ENABLED=false
TELEGRAM_BOT_TOKEN=
TELEGRAM_CHAT_ID=

DISCORD_ENABLED=false
DISCORD_WEBHOOK_URL=

# --- endpoints (rarely changed) ---------------------------------------------
CLOB_HOST=https://clob.polymarket.com
GAMMA_HOST=https://gamma-api.polymarket.com
DATA_API_HOST=https://data-api.polymarket.com
CLOB_WS_URL=wss://ws-subscriptions-clob.polymarket.com/ws
HTTP_TIMEOUT=10
